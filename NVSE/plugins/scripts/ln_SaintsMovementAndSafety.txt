; ToggleDisableSaves 0/1
; EnableFastTravel 0/1

; UDFs

ref TickSavingUDF = CompileScript "SaintsMovementAndSafety\SMAS_TickSaving.gek"
ref OnFastTravelMenuOpenUDF = CompileScript "SaintsMovementAndSafety\SMAS_OnFastTravelMenuOpen.gek"
ref OnFastTravelMenuCloseUDF = CompileScript "SaintsMovementAndSafety\SMAS_OnFastTravelMenuClose.gek"
ref OnItemActivateUDF = CompileScript "SaintsMovementAndSafety\SMAS_OnItemActivate.gek"
ref OnSleepUDF = CompileScript "SaintsMovementAndSafety\SMAS_OnSleep.gek"

; Tick

SetGameMainLoopCallback TickSavingUDF 1 60 1

; Menu

SetOnMenuOpenEventHandler OnFastTravelMenuOpenUDF 1 1023
SetOnMenuCloseEventHandler OnFastTravelMenuCloseUDF 1 1023

; Sleep/Wait

SetJohnnyOnSleepWaitEventHandler 1 OnSleepUDF 0 2

; Items

ref rItemFormList = TempCloneForm HelpManual
array_var aItemRefs = GetKeywordForms "SaveItems"

ListClear rItemFormList
ListAddArray rItemFormList aItemRefs

SetOnUseAidItemEventHandler (begin function { ref rItem } 
	Call OnItemActivateUDF rItem
end) 1 rItemFormList

; Log

Print "Set up event handlers for Saint's Movement & Safety."
